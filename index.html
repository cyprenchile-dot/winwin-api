function updateUI() { let totalDaily = 0; let totalBox = 0; // Aquí calcularemos el acumulado físico real de la flota let totalPrizes = 0; let onlineCount = 0; let wifiSum = 0; let todayStr = new Date().toISOString().slice(0, 10); machines.forEach(m => { let prevSales = previousSalesState[m.mac] || 0; let currentSales = m.sales || 0; // Efecto de sonido y luz verde en tiempo real if (currentSales > prevSales) { let diff = currentSales - prevSales; let coinsCount = Math.floor(diff / 100); for (let i = 0; i < coinsCount; i++) { setTimeout(() => playCoinDropSound(), i * 200); } triggerGreenPulse(m.mac); } previousSalesState[m.mac] = currentSales; if (!m.dailyLogs) m.dailyLogs = {}; if (m.lastSalesCount === undefined) m.lastSalesCount = currentSales; if (currentSales > m.lastSalesCount) { let diff = currentSales - m.lastSalesCount; m.dailyLogs[todayStr] = (m.dailyLogs[todayStr] || 0) + diff; m.lastSalesCount = currentSales; } else if (currentSales < m.lastSalesCount) { m.lastSalesCount = currentSales; } let isOnline = m.is_online !== false; if (isOnline) onlineCount++; // 1. Venta estricta del día actual por máquina let machineTodaySales = (m.dailyLogs && m.dailyLogs[todayStr]) ? m.dailyLogs[todayStr] : 0; totalDaily += machineTodaySales; // 2. DINERO REAL EN CAJA FÍSICA BASADO EN CONTADORES (Caja Global) let cInit = m.coinInitial !== undefined ? m.coinInitial : 3768; let telemetryPulses = Math.floor((m.sales || 0) / 100); let cFinal = (m.coinManualBase !== undefined) ? (m.coinManualBase + telemetryPulses) : (cInit + telemetryPulses); let cDelta = Math.max(0, cFinal - cInit); let dineroRegistradoPorContador = cDelta * 100; // Sumatoria absoluta para la Caja Global (Físico) totalBox += dineroRegistradoPorContador; let pDelta = m.prizes || 0; totalPrizes += pDelta; wifiSum += (m.wifi || -50); }); let avgWifi = machines.length > 0 ? Math.round(wifiSum / machines.length) : 0; let efficiency = totalPrizes > 0 ? Math.min(100, Math.round((totalPrizes * 500 / (totalDaily || 1)) * 100)) : 0; // Renderizado de métricas globales blindadas document.getElementById('daily-sales').innerText = `$${totalDaily.toLocaleString()} CLP`; document.getElementById('box-total').innerText = `$${totalBox.toLocaleString()} CLP`; document.getElementById('total-prizes').innerText = `${totalPrizes} un.`; document.getElementById('active-terminals').innerText = onlineCount; document.getElementById('wifi-signal').innerText = `${avgWifi} dBm`; document.getElementById('fleet-efficiency').innerText = `${efficiency}%`; salesChart.data.datasets[0].data[6] = totalDaily; salesChart.update(); renderMachineBoxes(machines); renderLedger(); renderAuditLog(); updateFleetMapMarkers(machines); }
