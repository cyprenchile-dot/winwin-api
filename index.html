async function resetMachineBox(index) { let m = machines[index]; let operador = prompt(`Retiro para "${m.name}". Ingresa quién realiza el retiro (Piero o Daniel):`, "Daniel"); if (!operador) return; let observacion = prompt("Ingresa alguna observación o novedad de la máquina (opcional):", "Sin novedad"); let montoRetirado = m.box; let premiosRetirados = m.prizes || 0; if (!m.withdrawalHistory) m.withdrawalHistory = []; m.withdrawalHistory.unshift({ date: new Date().toLocaleString(), operator: operador, amount: montoRetirado, prizesCount: premiosRetirados, note: observacion || "Sin observaciones" }); // Reiniciamos ambos contadores a cero para el nuevo ciclo m.box = 0; m.prizes = 0; await saveStateAndSync(); updateUI(); alert(`✅ Retiro registrado por ${operador}\nMonto: $${montoRetirado.toLocaleString()} CLP\nPremios reiniciados a 0.`); }
